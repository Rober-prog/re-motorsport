const slugs = [
  'mecanica',
  'frenos',
  'suspension',
  'filtros',
  'sistema-electrico',
  'iluminacion',
  'climatizacion',
  'transmision',
  'escape',

test('todas las rutas prerenderizadas contienen español y datos comerciales prudentes', async ({
  request,
}) => {
  for (const route of [
    '',
    '/empresa',
    '/catalogo',
    '/tiendas',
    '/contacto',
    '/profesionales',
    '/promociones',
    '/clasicos',
    '/aviso-legal',
    '/privacidad',
    '/cookies',
    ...slugs.map((s) => '/catalogo/' + s),
  ]) {
    for (const prefix of ['', '/es', '/ca']) {
      const response = await request.get(prefix + route || '/');
      expect(response.status()).toBe(200);
      const html = await response.text();
      expect(html).toContain('<h1');
      expect(html).toContain('<html lang="es"');
      expect(html).toContain('RE-Motorsport');
      expect(html).not.toMatch(
        /Martorell|rmartorell|B60408010|B66294414|B02658516|recambiosclasicos|wa\.me|mailto:/i,
      );
      expect(html).toContain('noindex,follow');
      expect(html).not.toContain('rel="canonical"');
      const schema = JSON.parse(
        html.match(/<script type="application\/ld\+json">([\s\S]*?)<\/script>/)![1],
      );
      const stores = schema['@graph'].filter(
        (v: Record<string, unknown>) => v['@type'] === 'AutoPartsStore',
      );
      expect(stores).toHaveLength(2);
      expect(JSON.stringify(stores)).not.toMatch(/Olesa|rating|taxID|openingHours/);
    }
  }
  expect(await (await request.get('/robots.txt')).text()).toContain('Disallow: /');
  expect((await request.get('/models/lamborghini-temerario.web.glb')).status()).toBe(200);
  expect((await request.get('/models/Lamborghini+Temerario.glb')).status()).toBe(404);
});

test('sin WebGL ofrece alternativas y selector de categorías', async ({ page }) => {
  await page.addInitScript(() => {
    const original = HTMLCanvasElement.prototype.getContext;
    HTMLCanvasElement.prototype.getContext = function (
      this: HTMLCanvasElement,
      type: string,
      ...args: unknown[]
    ) {
      if (type.includes('webgl')) return null;
      return Reflect.apply(original, this, [type, ...args]);
    } as typeof original;
  });
  await page.goto('/');
  await expect(page.getByText(/La vista 3D no está disponible/)).toBeVisible();
  await page.getByLabel('Explorar categorías', { exact: true }).selectOption('frenos');
  await expect(page.getByRole('dialog')).toContainText('Frenos');
  await page.keyboard.press('Escape');
  await expect(page.locator('#vehicle-popup')).toHaveCount(0);
});
test('visor 3D recuperado: carga, hotspot, giro, reinicio y pausa', async ({ page }) => {
  test.setTimeout(180000);
  await page.setViewportSize({ width: 1440, height: 1000 });
  const errors: string[] = [];
  page.on('pageerror', (e) => errors.push(e.message));
  await page.goto('/');
  await page.locator('[data-hotspot="mecanica"]').waitFor({ timeout: 120000 });
  await expect(page.locator('.vehicle-canvas canvas')).toBeVisible();
  await page.getByLabel('Explorar categorías', { exact: true }).selectOption('mecanica');
  await expect(page.getByRole('dialog')).toContainText('Motor y distribución');
  await page.keyboard.press('Escape');
  await page.getByRole('button', { name: 'Girar a la derecha', exact: true }).click();
  await page.getByRole('button', { name: 'Restablecer vista', exact: true }).click();
  await page.screenshot({ path: 'docs/previews/re-motorsport-desktop.png', fullPage: true });
  await page.getByRole('button', { name: 'Desactivar experiencia 3D', exact: true }).click();
  await expect(page.locator('.vehicle-canvas')).toHaveCount(0);
  await expect(page.getByRole('button', { name: 'Activar experiencia 3D' })).toBeVisible();
  expect(errors).toEqual([]);
});

test('contraste y semántica de las páginas principales', async ({ page }) => {
  test.setTimeout(180000);
  await page.emulateMedia({ reducedMotion: 'reduce' });
  for (const route of ['/', '/catalogo', '/contacto', '/tiendas', '/empresa', '/privacidad']) {
    await page.goto(route);
    const result = await new AxeBuilder({ page })
      .withTags(['wcag2a', 'wcag2aa', 'wcag21a', 'wcag21aa', 'wcag22aa'])
      .analyze();
    expect(
      result.violations.map((v) => ({ id: v.id, nodes: v.nodes.map((n) => n.target) })),
      route,
    ).toEqual([]);
  }
  await page.setViewportSize({ width: 390, height: 844 });
  await page.goto('/');
  await page.screenshot({ path: 'docs/previews/re-motorsport-mobile.png', fullPage: true });
});
